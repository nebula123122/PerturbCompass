#!/usr/bin/env python3
"""Create V38 source-only OOD clusters without reading target effect layers."""

from __future__ import annotations

import hashlib
import importlib.util
import json
from datetime import datetime, timezone
from pathlib import Path

import anndata as ad
import h5py
import numpy as np
import pandas as pd
import yaml
from sklearn.cluster import KMeans
from sklearn.decomposition import TruncatedSVD
from sklearn.preprocessing import normalize


ROOT = Path(__file__).resolve().parents[1]
CONFIG = ROOT / "configs/v38_source_defined_ood_predesign.yaml"
V36_CONFIG = ROOT / "configs/v36_cd4t_frozen.yaml"
FEASIBILITY_ADAPTATION = ROOT / "docs/audit/v38_cluster_feasibility_adaptation.json"


def load_module(name: str, path: Path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    assert spec.loader is not None
    spec.loader.exec_module(module)
    return module


V28 = load_module("v28_v38_prepare", ROOT / "scripts/run_v28_adamson_cross_study.py")
META = load_module("meta_v38_prepare", ROOT / "scripts/inspect_v36_cd4t_metadata.py")


def digest(path: Path) -> str:
    value = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(16 * 1024 * 1024), b""):
            value.update(block)
    return value.hexdigest()


def root_path(value: str) -> Path:
    return (ROOT / value).resolve()


def relabel_clusters(labels: np.ndarray, centers: np.ndarray) -> np.ndarray:
    ordering = sorted(range(len(centers)), key=lambda index: (float(centers[index, 0]), index))
    mapping = {old: new for new, old in enumerate(ordering)}
    return np.asarray([mapping[int(label)] for label in labels], dtype=int)


def cluster_counts(partition: pd.DataFrame, labels: np.ndarray, cluster_count: int) -> list[dict[str, int]]:
    rows = []
    for cluster in range(cluster_count):
        in_cluster = labels == cluster
        training_anchors = int((partition["role"].eq("anchor").to_numpy() & ~in_cluster).sum())
        test_targets = int((partition["role"].eq("test").to_numpy() & in_cluster).sum())
        rows.append({
            "cluster": cluster,
            "training_anchors_outside_cluster": training_anchors,
            "heldout_test_targets": test_targets,
        })
    return rows


def main() -> None:
    design = yaml.safe_load(CONFIG.read_text(encoding="utf-8"))
    v36 = yaml.safe_load(V36_CONFIG.read_text(encoding="utf-8"))
    adaptation = json.loads(FEASIBILITY_ADAPTATION.read_text(encoding="utf-8"))
    if adaptation.get("status") != "V38_SOURCE_ONLY_CLUSTER_FEASIBILITY_ADAPTATION_FROZEN_BEFORE_TARGET_NUMERIC_ACCESS":
        raise RuntimeError("V38 source-only cluster feasibility adaptation missing")
    partition_path = root_path(v36["frozen_inputs"]["shared_partition"])
    partition = pd.read_csv(partition_path).sort_values("target").reset_index(drop=True)
    partition["target"] = partition["target"].astype(str).str.upper()
    target_path = root_path(v36["target"]["path"])
    with h5py.File(target_path, "r") as handle:
        var = handle["var"]
        measured = META.read_frame_column(var, "gene_name") if "gene_name" in var else META.read_index(var)
    target_genes = {str(value).strip().upper() for value in measured if str(value).strip()}
    source_data = ad.read_h5ad(root_path(v36["source"]["h5ad"]))
    lookup = V28.gene_lookup(source_data)
    genes = sorted(set(lookup) & target_genes)
    columns = np.asarray([lookup[gene] for gene in genes], dtype=np.int64)
    source_targets, source_means, _, _ = V28.aggregate_source(
        source_data, columns, int(v36["source"]["minimum_cells_per_construct"])
    )
    del source_data
    source_map = {str(target).upper(): index for index, target in enumerate(source_targets)}
    source = np.stack([source_means[source_map[target]] for target in partition["target"]]).astype(np.float32)
    definition = design["ood_definition"]
    embedding = TruncatedSVD(
        n_components=int(definition["cluster_count"]) * 4,
        random_state=int(definition["svd_seed"]),
    ).fit_transform(source)
    embedding = normalize(embedding, norm="l2", axis=1)
    candidate_reports = []
    selected_count = None
    selected_labels = None
    counts = None
    for candidate_count in adaptation["candidate_cluster_counts_in_order"]:
        clusterer = KMeans(
            n_clusters=int(candidate_count),
            random_state=int(definition["kmeans_seed"]),
            n_init=int(definition["kmeans_n_init"]),
        ).fit(embedding)
        labels = relabel_clusters(clusterer.labels_, clusterer.cluster_centers_)
        candidate_counts = cluster_counts(partition, labels, int(candidate_count))
        feasible = all(
            row["training_anchors_outside_cluster"] >= int(definition["minimum_training_anchors_per_fold"])
            and row["heldout_test_targets"] >= int(definition["minimum_test_targets_per_cluster"])
            for row in candidate_counts
        )
        candidate_reports.append({"cluster_count": int(candidate_count), "feasible": feasible, "counts": candidate_counts})
        if feasible:
            selected_count = int(candidate_count)
            selected_labels = labels
            counts = candidate_counts
            break
    if selected_count is None or selected_labels is None or counts is None:
        raise RuntimeError("V38 no prespecified candidate cluster count meets the original size thresholds")
    partition["source_ood_cluster"] = selected_labels
    output = root_path(design["outputs"]["source_cluster_partition"])
    output.parent.mkdir(parents=True, exist_ok=True)
    partition.to_csv(output, index=False)
    manifest = {
        "status": "V38_SOURCE_CLUSTERS_FROZEN_BEFORE_V38_NUMERIC_ACCESS",
        "generated_at_utc": datetime.now(timezone.utc).isoformat(),
        "target_effect_layers_read": False,
        "cluster_feasibility_adaptation_sha256": digest(FEASIBILITY_ADAPTATION),
        "design_config_sha256": digest(CONFIG),
        "v36_config_sha256": digest(V36_CONFIG),
        "v36_partition_sha256": digest(partition_path),
        "source_cluster_partition_sha256": digest(output),
        "common_genes": len(genes),
        "cluster_counts": counts,
        "actual_cluster_count": selected_count,
        "candidate_cluster_reports_until_selection": candidate_reports,
        "preprocessing": {
            "svd_components": int(definition["cluster_count"]) * 4,
            "row_normalization": "L2",
            "deterministic_relabel": "centroid_first_component_then_original_label",
        },
    }
    manifest_path = root_path(design["outputs"]["pre_numeric_manifest"])
    manifest_path.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(manifest, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
