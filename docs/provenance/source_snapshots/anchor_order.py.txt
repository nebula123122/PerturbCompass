#!/usr/bin/env python3
"""Create and seal the V40 nested anchor order from source effects only."""

from __future__ import annotations

import hashlib
import importlib.util
import json
from datetime import datetime, timezone
from pathlib import Path

import anndata as ad
import h5py
import numpy as np
import pandas as pd
import yaml
from sklearn.decomposition import TruncatedSVD


ROOT = Path(__file__).resolve().parents[1]
CONFIG = ROOT / "configs/v40_anchor_budget_predesign.yaml"
V36_CONFIG = ROOT / "configs/v36_cd4t_frozen.yaml"
V36_SEAL = ROOT / "docs/audit/v36_pre_numeric_manifest.json"
RUNNER = ROOT / "scripts/run_v40_anchor_budget.py"
TESTS = ROOT / "tests/test_v40_anchor_budget.py"


def load_module(name: str, path: Path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    assert spec.loader is not None
    spec.loader.exec_module(module)
    return module


V28 = load_module("v28_v40_prepare", ROOT / "scripts/run_v28_adamson_cross_study.py")
META = load_module("meta_v40_prepare", ROOT / "scripts/inspect_v36_cd4t_metadata.py")


def digest(path: Path) -> str:
    value = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(16 * 1024 * 1024), b""):
            value.update(block)
    return value.hexdigest()


def root_path(value: str) -> Path:
    return (ROOT / value).resolve()


def farthest_first_order(features: np.ndarray, names: list[str]) -> list[int]:
    """Deterministic centroid start followed by maximin distance and lexical ties."""
    values = np.asarray(features, dtype=float)
    values = values / np.maximum(np.linalg.norm(values, axis=1, keepdims=True), 1.0e-12)
    centroid = values.mean(axis=0)
    first_distance = np.sum((values - centroid) ** 2, axis=1)
    first_candidates = np.flatnonzero(np.isclose(first_distance, first_distance.min(), rtol=1e-12, atol=1e-14))
    first = min(first_candidates, key=lambda index: names[int(index)])
    order = [int(first)]
    selected = np.zeros(len(values), dtype=bool)
    selected[first] = True
    minimum = np.sum((values - values[first]) ** 2, axis=1)
    while len(order) < len(values):
        maximum = minimum[~selected].max()
        candidates = np.flatnonzero((~selected) & np.isclose(minimum, maximum, rtol=1e-12, atol=1e-14))
        chosen = min(candidates, key=lambda index: names[int(index)])
        order.append(int(chosen))
        selected[chosen] = True
        minimum = np.minimum(minimum, np.sum((values - values[chosen]) ** 2, axis=1))
    return order


def main() -> None:
    if (ROOT / "docs/audit/v40_anchor_budget_summary.json").exists():
        raise RuntimeError("V40 outcome summary already exists")
    design = yaml.safe_load(CONFIG.read_text(encoding="utf-8"))
    v36 = yaml.safe_load(V36_CONFIG.read_text(encoding="utf-8"))
    partition = pd.read_csv(root_path(v36["frozen_inputs"]["shared_partition"])).sort_values("target").reset_index(drop=True)
    partition["target"] = partition["target"].astype(str).str.upper()
    anchors = partition.loc[partition["role"].eq(v36["partition"]["anchor_role"]), "target"].tolist()
    maximum_budget = max(map(int, design["anchor_design"]["budgets"]))
    if len(anchors) != maximum_budget:
        raise RuntimeError(f"expected {maximum_budget} V36 anchors, found {len(anchors)}")

    target_path = root_path(v36["target"]["path"])
    with h5py.File(target_path, "r") as handle:
        var = handle["var"]
        measured = META.read_frame_column(var, "gene_name") if "gene_name" in var else META.read_index(var)
    target_genes = {str(value).strip().upper() for value in measured if str(value).strip()}
    source_data = ad.read_h5ad(root_path(v36["source"]["h5ad"]))
    source_lookup = V28.gene_lookup(source_data)
    genes = sorted(set(source_lookup) & target_genes)
    source_columns = np.asarray([source_lookup[gene] for gene in genes], dtype=np.int64)
    source_targets, source_means, _, _ = V28.aggregate_source(
        source_data, source_columns, int(v36["source"]["minimum_cells_per_construct"])
    )
    del source_data
    source_map = {str(target).upper(): index for index, target in enumerate(source_targets)}
    missing = sorted(set(anchors) - set(source_map))
    if missing:
        raise RuntimeError(f"V36 anchors missing from source: {missing[:20]}")
    source = np.stack([source_means[source_map[target]] for target in anchors]).astype(np.float32)
    components = min(32, len(anchors) - 1, len(genes) - 1)
    embedded = TruncatedSVD(
        n_components=components, random_state=int(design["anchor_design"]["svd_seed"])
    ).fit_transform(source)
    order = farthest_first_order(embedded, anchors)
    ordered_targets = [anchors[index] for index in order]
    rows = []
    for rank, target in enumerate(ordered_targets, start=1):
        row = {"rank": rank, "target": target}
        for budget in map(int, design["anchor_design"]["budgets"]):
            row[f"in_budget_{budget}"] = rank <= budget
        rows.append(row)
    output = root_path(design["outputs"]["source_anchor_order"])
    output.parent.mkdir(parents=True, exist_ok=True)
    pd.DataFrame(rows).to_csv(output, index=False)
    if set(ordered_targets) != set(anchors) or len(set(ordered_targets)) != len(anchors):
        raise RuntimeError("V40 anchor order is not a permutation of the V36 anchor pool")

    manifest = {
        "status": "V40_SOURCE_ONLY_ANCHOR_ORDER_AND_IMPLEMENTATION_SEALED",
        "frozen_at_utc": datetime.now(timezone.utc).isoformat(),
        "design_was_frozen_before_v36_numeric_access": True,
        "order_was_computed_after_v36_results_but_without_reading_target_outcomes": True,
        "v40_target_outcomes_read": False,
        "config_sha256": digest(CONFIG),
        "v36_config_sha256": digest(V36_CONFIG),
        "v36_pre_numeric_manifest_sha256": digest(V36_SEAL),
        "v36_partition_sha256": digest(root_path(v36["frozen_inputs"]["shared_partition"])),
        "source_anchor_order_sha256": digest(output),
        "runner_sha256": digest(RUNNER),
        "tests_sha256": digest(TESTS),
        "common_genes": len(genes),
        "anchor_count": len(anchors),
        "budgets": list(map(int, design["anchor_design"]["budgets"])),
        "target_outcomes_used_for_order": False,
    }
    manifest_path = root_path(design["outputs"]["pre_numeric_manifest"])
    manifest_path.parent.mkdir(parents=True, exist_ok=True)
    manifest_path.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(manifest, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
